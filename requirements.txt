# --- CUDA 12 stack ----------------------------------------------------------
# torch 2.10.0 is the LAST PyPI release built on CUDA 12 (12.8); 2.11 onwards
# is CUDA 13. The speech engine (CTranslate2 4.8.2, below) is a CUDA 12 build
# that needs cuBLAS 12 and cuDNN 9 at run time - which torch 2.10.0 brings as
# its own dependencies (nvidia-cublas-cu12, nvidia-cudnn-cu12) on Linux, and
# duet/perception/cuda.py makes visible. One CUDA runtime, shipped by pip, so
# only the machine's driver matters, not its installed toolkit.
torch==2.10.0
torchvision==0.25.0

# --- speech -----------------------------------------------------------------
# faster-whisper: CTranslate2 inference, word-level probabilities (what M5's
# confidence gate is built on), and an MP3/WAV decoder (PyAV) that needs no
# system ffmpeg.
faster-whisper==1.2.1
ctranslate2==4.8.2

# --- vision -----------------------------------------------------------------
# transformers + accelerate: the vision-language model (device_map="auto"
# requires accelerate). sentence-transformers: CLIP frame embeddings.
transformers==5.4.0
accelerate==1.15.0

huggingface-hub==1.8.0

# --- shared -----------------------------------------------------------------
